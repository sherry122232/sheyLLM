```python
class MultiHeadAttention(nn.Module):
    """多头注意力机制"""
    def __init__(self, d_model: int, num_heads: int, dropout: float = 0.1):
        super().__init__()
        assert d_model % num_heads == 0, "d_model must be divisible by num_heads"
        
        self.d_model = d_model
        self.num_heads = num_heads
        self.d_k = d_model // num_heads  # 每个头的维度
        
        # Q, K, V 的线性变换层
        self.W_q = nn.Linear(d_model, d_model)
        self.W_k = nn.Linear(d_model, d_model)
        self.W_v = nn.Linear(d_model, d_model)
        self.W_o = nn.Linear(d_model, d_model)
        
        self.dropout = nn.Dropout(dropout)
        
    def forward(self, query, key, value, mask=None):
        """
        query: [batch, seq_q, d_model]
        key:   [batch, seq_k, d_model]
        value: [batch, seq_v, d_model]
        mask:  [batch, seq_q, seq_k] or [batch, num_heads, seq_q, seq_k]
        """
        batch_size = query.size(0)
        
        # 1. 线性变换并拆分为多头
        Q = self.W_q(query).view(batch_size, -1, self.num_heads, self.d_k).transpose(1, 2)
        K = self.W_k(key).view(batch_size, -1, self.num_heads, self.d_k).transpose(1, 2)
        V = self.W_v(value).view(batch_size, -1, self.num_heads, self.d_k).transpose(1, 2)
        # Q, K, V: [batch, num_heads, seq_len, d_k]
        
        # 2. 计算注意力分数
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k)
        # scores: [batch, num_heads, seq_q, seq_k]
        
        # 3. 应用 mask（如果提供）
        if mask is not None:
            scores = scores.masked_fill(mask == 0, float('-inf'))
        
        # 4. Softmax + Dropout
        attn = F.softmax(scores, dim=-1)
        attn = self.dropout(attn)
        
        # 5. 应用注意力权重到 V
        context = torch.matmul(attn, V)
        # context: [batch, num_heads, seq_q, d_k]
        
        # 6. 合并多头并输出
        context = context.transpose(1, 2).contiguous().view(batch_size, -1, self.d_model)
        output = self.W_o(context)
        
        return output, attn

```

# GQA
```python
class GroupedQueryAttention(nn.Module):
    def __init__(self, d_model, num_heads, num_kv_heads):
        super().__init__()
        self.num_heads = num_heads
        self.num_kv_heads = num_kv_heads
        self.num_groups = num_heads // num_kv_heads
        
        self.d_k = d_model // num_heads
        
        # Q: 多头，K/V: 分组头
        self.W_q = nn.Linear(d_model, d_model)
        self.W_k = nn.Linear(d_model, num_kv_heads * self.d_k)
        self.W_v = nn.Linear(d_model, num_kv_heads * self.d_k)
        self.W_o = nn.Linear(d_model, d_model)
    
    def forward(self, q, k, v, mask=None):
        batch_size = q.size(0)
        
        # Q: [batch, seq_q, num_heads, d_k]
        Q = self.W_q(q).view(batch_size, -1, self.num_heads, self.d_k)
        
        # K, V: [batch, seq_k, num_kv_heads, d_k]
        K = self.W_k(k).view(batch_size, -1, self.num_kv_heads, self.d_k)
        V = self.W_v(v).view(batch_size, -1, self.num_kv_heads, self.d_k)
        
        # 扩展 K, V 到与 Q 相同的头数 (重复分组)
        K = K.repeat_interleave(self.num_groups, dim=2)  # [batch, seq_k, num_heads, d_k]
        V = V.repeat_interleave(self.num_groups, dim=2)
        
        # 标准注意力计算
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k)
        attn = F.softmax(scores, dim=-1)
        output = torch.matmul(attn, V)
        
        return self.W_o(output.flatten(2))
```
# MQA
```python
class MultiQueryAttention(nn.Module):
    def __init__(self, d_model, num_heads):
        super().__init__()
        self.num_heads = num_heads
        self.d_k = d_model // num_heads
        
        # Q: 多头，K/V: 单头
        self.W_q = nn.Linear(d_model, d_model)           # [d_model, d_model]
        self.W_k = nn.Linear(d_model, self.d_k)          # [d_model, d_k]
        self.W_v = nn.Linear(d_model, self.d_k)          # [d_model, d_k]
        self.W_o = nn.Linear(d_model, d_model)
    
    def forward(self, q, k, v, mask=None):
        batch_size = q.size(0)
        
        # Q: [batch, seq_q, num_heads, d_k]
        Q = self.W_q(q).view(batch_size, -1, self.num_heads, self.d_k)
        
        # K, V: [batch, seq_k, d_k] (单头)
        K = self.W_k(k)  # [batch, seq_k, d_k]
        V = self.W_v(v)  # [batch, seq_k, d_k]
        
        # 计算注意力 (需要广播)
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k)
        # scores: [batch, num_heads, seq_q, seq_k]
        
        attn = F.softmax(scores, dim=-1)
        output = torch.matmul(attn, V.unsqueeze(1))  # 广播 V
        
        return self.W_o(output.flatten(2))
```

# MLA
```python
class MultiHeadLatentAttention(nn.Module):
    def __init__(self, d_model, num_heads, latent_rank):
        super().__init__()
        self.num_heads = num_heads
        self.d_k = d_model // num_heads
        self.latent_rank = latent_rank  # 压缩维度，通常 d_model/4
        
        # Q 的压缩投影 (可选)
        self.W_dq = nn.Linear(d_model, latent_rank)
        self.W_uq = nn.Linear(latent_rank, d_model)
        
        # KV 的压缩投影 (核心)
        self.W_down_kv = nn.Linear(d_model, latent_rank)
        self.W_up_k = nn.Linear(latent_rank, d_model)
        self.W_up_v = nn.Linear(latent_rank, d_model)
        
        self.W_o = nn.Linear(d_model, d_model)
    
    def forward(self, q, k, v, latent_cache=None):
        batch_size = q.size(0)
        
        # 压缩 K, V 到潜在空间
        latent_kv = self.W_down_kv(k)  # [batch, seq, latent_rank]
        
        if latent_cache is not None:
            # 推理时：只缓存潜在向量
            latent_kv = torch.cat([latent_cache, latent_kv], dim=1)
        
        # 解压缩到完整 K, V
        K = self.W_up_k(latent_kv).view(batch_size, -1, self.num_heads, self.d_k)
        V = self.W_up_v(latent_kv).view(batch_size, -1, self.num_heads, self.d_k)
        
        # Q 也做类似处理
        latent_q = self.W_dq(q)
        Q = self.W_uq(latent_q).view(batch_size, -1, self.num_heads, self.d_k)
        
        # 标准注意力计算
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k)
        attn = F.softmax(scores, dim=-1)
        output = torch.matmul(attn, V)
        
        return self.W_o(output.flatten(2)), latent_kv
```
